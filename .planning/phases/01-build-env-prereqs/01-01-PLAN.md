    ```powershell
      $lines += "状态: FAIL"
    ```

    **步骤 2b: .NET SDK 版本验证**

    ```powershell
    $dotnetFile = Join-Path $outDir 'dotnet-sdk-check.txt'

    $dotnetVersion = & dotnet --version 2>&1
    $globalJson = Get-Content (Join-Path $root 'global.json') -Raw | ConvertFrom-Json
    $expectedVersion = $globalJson.sdk.version
    $rollForward = $globalJson.sdk.rollForward

    $dotnetLines = @(
      '========================================',
      '.NET SDK 版本验证',
      '========================================',
      "检查时间: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')",
      '----------------------------------------',
      "来源: global.json",
      "指定版本 (sdk.version): $expectedVersion",
      "rollForward 策略: $rollForward",
      '----------------------------------------',
      "实际 dotnet --version: $dotnetVersion",
      ''
    )

    # 判断兼容性: rollForward: latestFeature 允许在同一 feature band 内使用任何版本
    # 即 10.0.x 中 x >= 指定值? 实际上 latestFeature 允许的是 feature band 内的最大值，
    # 但如果有比 10.0.201 更高的 patch 版本则使用更高的。如果没有，就回退到可用的最低兼容版本。
    # 对于 latestFeature: 要求 major.minor 匹配（10.0），feature band 内任意 patch 可用。
    $actualMajorMinor = ($dotnetVersion -split '\.' | Select-Object -First 2) -join '.'
    $expectedMajorMinor = ($expectedVersion -split '\.' | Select-Object -First 2) -join '.'
    $compatible = $actualMajorMinor -eq $expectedMajorMinor

    $dotnetLines += "major.minor 匹配: $($compatible ? '是' : '否')"
    $dotnetLines += "实际: $dotnetVersion (major.minor = $actualMajorMinor)"
    $dotnetLines += "期望: $expectedVersion 或同 feature band 任意版本 (major.minor = $expectedMajorMinor)"
    if ($compatible) {
      $dotnetLines += "状态: PASS (rollForward: $rollForward 允许使用 $dotnetVersion)"
    } else {
      $dotnetLines += "状态: FAIL - .NET SDK 版本不兼容"
      $dotnetLines += "修复指引: 安装 .NET SDK $expectedVersion 或同 feature band 的版本"

    Write-Output ".NET SDK 验证结果已保存到: $dotnetFile"
    ```
